Deploy script - save as /home/allianceserver/myauth/push_plugin.sh

    #!/bin/bash
    set -e

    PROJECT="$1"

    if [ -z "$PROJECT" ]; then
      echo "Usage: $0 <project-name>"
      exit 1
    fi

    WIN_PATH="/mnt/c/Users/agolf/PycharmProjects/$PROJECT/"
    LINUX_PATH="/home/allianceserver/myauth/$PROJECT"
    VENV_PATH="/home/allianceserver/venv/auth/bin/activate"
    AUTH_PATH="/home/allianceserver/myauth"

    rsync -av --delete "$WIN_PATH" "$LINUX_PATH/"

    cd "$AUTH_PATH"
    source "$VENV_PATH"
    pip install -e "$LINUX_PATH"
    python manage.py migrate
    python manage.py collectstatic --noinput
    supervisorctl restart myauth:

Run it:

    sudo su - allianceserver
